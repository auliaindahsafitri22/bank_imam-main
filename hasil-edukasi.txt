31:  // Fungsi helper untuk mengambil domain utama dari URL
32:  private function getHostFromUrl($url)
33:  {
34:  if (empty($url) || trim($url) === '') {
35:  return 'EcoBuana Edukasi';
36:  }
37:  if (!preg_match('~^(?:f|ht)tps?://~i', $url)) {
38:  $url = 'https://' . ltrim($url, '/');
39:  }
40:  $parsedUrl = parse_url($url);
41:  $host = $parsedUrl['host'] ?? 'EcoBuana Edukasi';
42:  if (empty($host) || $host === 'undefined') {
43:  return 'EcoBuana Edukasi';
44:  }
45:  return preg_replace('/^www\./', '', $host);
46:  }
